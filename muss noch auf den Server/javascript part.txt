function toggleView(viewType) {
  const grid = document.getElementById('cards-grid');
  if (viewType === 'list') {
    grid.classList.add('list-view');
  } else {
    grid.classList.remove('list-view');
  }
}

Use this function to clone the template and populate it with data.

function addEvent(tag, imageSrc, title, subtitle, link) {
  const grid = document.getElementById('cards-grid');
  const template = document.getElementById('card-template');
  
  // Clone template
  const clone = template.content.cloneNode(true);
  
  // Fill data
  clone.querySelector('.card-tag').textContent = tag;
  clone.querySelector('.card-image img').src = imageSrc;
  clone.querySelector('.card-title').textContent = title;
  clone.querySelector('.card-subtitle').textContent = subtitle;
  clone.querySelector('.card-link').href = link;
  
  // Append to grid
  grid.appendChild(clone);
}

Like button function

/**
 * Toggles the like state and updates the count.
 * @param {HTMLElement} element - The card-footer element that was clicked.
 */
function toggleLike(element) {
  const countElement = element.querySelector('.card-count');
  const iconElement = element.querySelector('.card-icon');
  
  // Get current count
  let count = parseInt(countElement.textContent);

  // Check if already liked (toggle class)
  const isLiked = element.classList.toggle('liked');

  // Update count and visual state
  if (isLiked) {
    count++;
    iconElement.style.color = 'red'; // Visual indicator
  } else {
    count--;
    iconElement.style.color = 'inherit'; // Reset color
  }

  // Update display
  countElement.textContent = count;
}
